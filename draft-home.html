import numpy as np import pandas as pd class DataPipeline: def __init__(self, target_column: str): self.target_column = target_column self.mean_values = {} def fit(self, df: pd.DataFrame): """Calculate and store feature metrics from training data.""" numeric_cols = df.select_dtypes(include=[np.number]).columns for col in numeric_cols: if col != self.target_column: self.mean_values[col] = df[col].mean() return self def transform(self, df: pd.DataFrame) -> pd.DataFrame: """Apply missing value imputation and feature scaling.""" df_transformed = df.copy() # Impute missing numeric values using fitted means for col, mean_val in self.mean_values.items(): if col in df_transformed.columns: df_transformed[col] = df_transformed[col].fillna(mean_val) # Standardize features (Z-score normalization) for col in self.mean_values.keys(): if col in df_transformed.columns: std = df_transformed[col].std() if std != 0: df_transformed[col] = (df_transformed[col] - self.mean_values[col]) / std return df_transformed def fit_transform(self, df: pd.DataFrame) -> pd.DataFrame: """Fit parameters and transform input data in a single step.""" return self.fit(df).transform(df) # Example Execution if __name__ == "__main__": # Sample Dataset Construction raw_data = { 'feature_a': [10.0, 12.0, np.nan, 14.0, 15.0], 'feature_b': [100, 200, 150, np.nan, 300], 'target': [0, 1, 0, 1, 0] } df_raw = pd.DataFrame(raw_data) # Process Pipeline pipeline = DataPipeline(target_column='target') df_processed = pipeline.fit_transform(df_raw) print("Processed Dataset:") print(df_processed)
